        if (!rawName) return '';
        return rawName
            .replace(/[的\-_\s]*(抖音|douyin|Dy|DY|TikTok)[\s\-_]*$/i, '')
            .replace(/^[\s\-_\|]+|[\s\-_\|]+$/g, '')
            .trim();
    }

    // --- 核心修复：页面类型检测 ---
    function detectPageType() {
        const url = window.location.href;
        const title = document.title.toLowerCase();

        // 精确判断自己页面
        const isSelfPage =
            url.includes('/user/self') ||
            url.includes('showTab=post') ||
            title.includes('我的主页') ||
            title.includes('个人主页');

        // 判断他人页面
        const isOthersPage = url.includes('/user/') && !url.includes('/user/self');

        // 设置页面类型
        if (isSelfPage) {
            userData.isOwnPage = true;
            console.log('[提取器] 检测到: 自己页面');
        } else if (isOthersPage) {
            userData.isOwnPage = false;
            console.log('[提取器] 检测到: 他人页面');
        } else {
            // 默认根据数据判断
            userData.isOwnPage = userData.secUid && userData.secUid.length > 20;
        }

        return userData.isOwnPage;
    }

    // --- 新增：增强的自动触发函数（模拟点击） ---
    function enhancedAutoTrigger() {
        if (autoClickAttempted || userData.secUid) {
            console.log('[提取器] 已尝试过或已有数据，跳过自动触发');
            return;
        }
        console.log('[提取器] 开始增强版自动获取...');
        autoClickAttempted = true;

        // 策略：优先寻找并点击"关注列表"，其次是"关注"按钮
        const selectors = [
            '[data-e2e="user-follow-list"]', // 关注列表按钮
            '[data-e2e="user-follow"]',      // 关注按钮
            '.dy-account-follow',            // 关注按钮的类
            'button:contains("关注")',        // 包含"关注"文字的按钮
            'div:contains("关注列表")'        // 包含"关注列表"文字的div
        ];

        let found = false;
        for (let selector of selectors) {
            const elements = document.querySelectorAll(selector);
            if (elements.length > 0) {
                console.log(`[提取器] 找到元素 ${selector}，尝试触发`);
                elements[0].click();
                found = true;
                // 点击后短暂延迟，给网络请求留出时间
                setTimeout(() => {
                    console.log('[提取器] 自动触发点击完成');
                }, 800);
                break;
            }
        }

        if (!found) {
            console.log('[提取器] 未找到合适的触发元素，将在面板显示手动指引');
        }
    }

    // --- 核心修复：网络请求监听 ---
    function setupNetworkListener() {
        const originalOpen = XMLHttpRequest.prototype.open;
        const originalSend = XMLHttpRequest.prototype.send;
        const originalFetch = window.fetch;

        XMLHttpRequest.prototype.open = function(method, url) {
            this._requestUrl = url;
            this._requestMethod = method;
            return originalOpen.apply(this, arguments);
        };

        XMLHttpRequest.prototype.send = function(body) {
            const requestUrl = this._requestUrl;
            if (!requestUrl) return originalSend.apply(this, arguments);

            // 1. 监听他人作品列表请求 - 获取博主UID (author_user_id)
            if (requestUrl.includes('/web/aweme/post')) {
                const originalOnReadyStateChange = this.onreadystatechange;
                this.onreadystatechange = function() {
                    if (this.readyState === 4 && this.status === 200) {
                        try {
                            const response = JSON.parse(this.responseText);
                            if (response && response.aweme_list && response.aweme_list.length > 0) {
                                const awemeItem = response.aweme_list[0];

                                // 只在他人页面时更新博主信息
                                if (!userData.isOwnPage) {
                                    // 关键修复：从author_user_id字段获取博主UID
                                    if (awemeItem.author_user_id) {
                                        userData.uid = awemeItem.author_user_id.toString();
                                        console.log('[提取器] 提取到博主UID (author_user_id):', userData.uid);
                                    }

                                    // 从author对象获取其他信息
                                    if (awemeItem.author) {
                                        userData.nickname = cleanNickname(awemeItem.author.nickname || '');
                                        userData.secUid = awemeItem.author.sec_uid || awemeItem.author.sec_user_id || '';
                                    }

                                    userData.lastUpdated = new Date().toLocaleTimeString();
                                    updateDisplay();
                                }
                            }
                        } catch (e) {
                            console.error('[提取器] 解析作品列表失败:', e);
                        }
                    }
                    if (originalOnReadyStateChange) {
                        return originalOnReadyStateChange.apply(this, arguments);
                    }
                };
            }

            // 2. 监听关注列表请求 - 获取自己的sec_user_id（修复：从 data.user.sec_uid 提取）
            if (requestUrl.includes('/web/user/following/list') || requestUrl.includes('/aweme/v1/web/user/following/list')) {
                console.log('[提取器] 检测到关注列表请求:', requestUrl);

                // 从URL提取sec_user_id（优先）
                try {
                    const urlObj = new URL(requestUrl, window.location.origin);
                    const secUidFromUrl = urlObj.searchParams.get('sec_user_id');
                    if (secUidFromUrl && secUidFromUrl.length > 10) {
                        userData.secUid = secUidFromUrl;
                        userData.isOwnPage = true;
                        console.log('[提取器] 从URL参数提取到SecUserID:', userData.secUid);
                    }
                } catch (e) { console.warn('[提取器] URL解析失败:', e); }

                // 从响应中提取（确保正确路径）
                const originalOnReadyStateChange = this.onreadystatechange;
                this.onreadystatechange = function() {
                    if (this.readyState === 4 && this.status === 200) {
                        try {
                            const response = JSON.parse(this.responseText);
                            let extractedSecUid = '';

                            // ✅ 正确路径：从 data.user.sec_uid 获取当前登录用户的 SecUserID
                            if (response?.data?.user?.sec_uid) {
                                extractedSecUid = response.data.user.sec_uid;
                            }
                            // 备用路径：某些接口可能直接返回 user 对象
                            else if (response?.user?.sec_uid) {
                                extractedSecUid = response.user.sec_uid;
                            }

                            if (extractedSecUid && extractedSecUid.length > 10) {
                                userData.secUid = extractedSecUid;
                                userData.isOwnPage = true;
                                userData.lastUpdated = new Date().toLocaleTimeString() + ' (自动获取)';
                                console.log('[提取器] 从响应JSON提取到SecUserID:', userData.secUid);
                                updateDisplay();
                                if (typeof GM_notification !== 'undefined') {
                                    GM_notification({
                                        text: `SecUserID 获取成功！`,
                                        title: '抖音信息提取器',
                                        timeout: 3000
                                    });
                                }
                            }
                        } catch (parseError) {
                            console.warn('[提取器] 解析关注列表响应失败:', parseError);
                        }
                    }
                    if (originalOnReadyStateChange) {
                        return originalOnReadyStateChange.apply(this, arguments);
                    }
                };
            }

            return originalSend.apply(this, arguments);
        };

        // 监听Fetch请求
        window.fetch = function(input, init) {
            const requestUrl = typeof input === 'string' ? input : input.url;
            if (!requestUrl) return originalFetch.apply(this, arguments);

            // 监听关注列表请求
            if (requestUrl.includes('/web/user/following/list') || requestUrl.includes('/aweme/v1/web/user/following/list')) {
                console.log('[提取器] Fetch检测到关注列表请求');

                try {
                    const urlObj = new URL(requestUrl, window.location.origin);
                    const secUidFromUrl = urlObj.searchParams.get('sec_user_id');

                    if (secUidFromUrl) {
                        userData.secUid = secUidFromUrl;
                        userData.isOwnPage = true;
                        userData.lastUpdated = new Date().toLocaleTimeString();
                        console.log('[提取器] Fetch提取到SecUserID:', userData.secUid);
                        updateDisplay();
                    }
                } catch (e) {
                    console.error('[提取器] Fetch URL解析失败:', e);
                }
            }

            return originalFetch.apply(this, arguments);
        };
    }

    // --- 单个复制功能 ---
    async function handleCopy(event) {
        const button = event.currentTarget;
        const text = button.getAttribute('data-text');

        if (!text) {
            button.innerHTML = '<span>❌ 无内容</span>';
            setTimeout(() => button.innerHTML = '<span>📋 复制</span>', 2000);
            return;
        }

        try {
            await navigator.clipboard.writeText(text);
            GM_setClipboard(text, 'text');

            button.innerHTML = '<span>✅ 已复制</span>';
            button.classList.add('copied');

            setTimeout(() => {
                button.innerHTML = '<span>📋 复制</span>';
                button.classList.remove('copied');
            }, 2000);

        } catch (err) {
            button.innerHTML = '<span>❌ 失败</span>';
            setTimeout(() => button.innerHTML = '<span>📋 复制</span>', 2000);
        }
    }
